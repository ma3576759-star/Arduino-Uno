💻 كود الأردوينو

laser_counter.ino — Arduino IDE
// ============================================
//  Laser Visitor Counter v2
// ============================================
#include <Wire.h>
#include <Adafruit_GFX.h>
#include <Adafruit_SSD1306.h>

#define W 128
#define H 64
Adafruit_SSD1306 oled(W, H, &Wire, -1);

const int LASER = 9;   // TX  ← D9
const int SENSOR = 2;  // RX  ← D2
const int BUZZ  = 8;   // Buzzer ← D8

int count = 0;

void setup() {
  pinMode(LASER, OUTPUT);
  pinMode(SENSOR, INPUT);
  pinMode(BUZZ, OUTPUT);
  digitalWrite(LASER, HIGH);      // شعاع شغال

  oled.begin(SSD1306_SWITCHCAPVCC, 0x3C);
  oled.clearDisplay();
  draw();
}

void loop() {
  if (digitalRead(SENSOR) == LOW) {   // الزائر قطع الشعاع
    count++;
    tone(BUZZ, 1000, 200);          // بيب!
    draw();
    delay(800);                  // منع العد المزدوج
  }
}

void draw() {
  oled.clearDisplay();
  oled.setTextColor(WHITE);
  oled.setTextSize(1);
  oled.setCursor(30, 6);
  oled.print("VISITORS");
  oled.setTextSize(3);
  oled.setCursor(40, 30);
  oled.print(count);
  oled.display();
}

🔄 طريقة العمل

🔦 إرسال

الـ KY-008 يطلق شعاع أحمر مستمر ناحية المستقبل في الطرف الآخر.

👁️ استقبال

المستقبل يستقبل الشعاع ويطلع إشارة HIGH ثابتة للأردوينو.

🚗 قطع

زائر يعدّي → الشعاع ينقطع → الإشارة تنزل LOW فجأة.

🧠 عدّ

الأردوينو يكتشف النزول ويزود العداد ويحدّث الشاشة.

📢 تنبيه

البازر يبيب والشاشة تعرض العدد الجديد، والتأخير يمنع التكرار.
